-- ============================================================================
-- ACTIVIDAD: MODELO DE BASE DE DATOS - GAS LICUADO "EL VOLCÁN" (NORMALIZADO)
-- Gestor: MySQL / MariaDB (XAMPP)
-- ============================================================================

DROP DATABASE IF EXISTS el_volcan_db;
CREATE DATABASE el_volcan_db CHARACTER SET utf8mb4 COLLATE utf8mb4_spanish_ci;
USE el_volcan_db;

-- ----------------------------------------------------------------------------
-- 1. TABLAS NORMALIZADAS DE UBICACIÓN GEOGRÁFICA
-- ----------------------------------------------------------------------------
CREATE TABLE regiones (
    id_region INT AUTO_INCREMENT PRIMARY KEY,
    nombre_region VARCHAR(100) NOT NULL UNIQUE
) ENGINE=InnoDB;

CREATE TABLE comunas (
    id_comuna INT AUTO_INCREMENT PRIMARY KEY,
    id_region INT NOT NULL,
    nombre_comuna VARCHAR(100) NOT NULL,
    CONSTRAINT fk_comunas_regiones FOREIGN KEY (id_region) 
        REFERENCES regiones(id_region) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 2. TABLA: ROLES
-- ----------------------------------------------------------------------------
CREATE TABLE roles (
    id_rol INT AUTO_INCREMENT PRIMARY KEY,
    nombre_rol VARCHAR(50) NOT NULL UNIQUE,
    descripcion_permisos VARCHAR(255) NOT NULL
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 3. TABLA: USUARIOS
-- ----------------------------------------------------------------------------
CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    id_rol INT NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    primer_nombre VARCHAR(50) NOT NULL,
    segundo_nombre VARCHAR(50) NULL,
    apellido_paterno VARCHAR(50) NOT NULL,
    apellido_materno VARCHAR(50) NULL,
    fecha_nacimiento DATE NOT NULL,
    cuenta_activa BOOLEAN DEFAULT TRUE,
    fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_usuarios_roles FOREIGN KEY (id_rol) 
        REFERENCES roles(id_rol) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 4. TABLA: DIRECCIONES_USUARIO (Calle y número separados + Normalización Comuna)
-- ----------------------------------------------------------------------------
CREATE TABLE direcciones_usuario (
    id_direccion INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    id_comuna INT NOT NULL,
    nombre_calle VARCHAR(120) NOT NULL,
    numero_calle VARCHAR(20) NOT NULL,
    departamento_o_bloque VARCHAR(50) NULL,
    villa_o_poblacion VARCHAR(100) NULL,
    es_direccion_actual BOOLEAN DEFAULT TRUE,
    fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_direcciones_usuarios FOREIGN KEY (id_usuario) 
        REFERENCES usuarios(id_usuario) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_direcciones_comunas FOREIGN KEY (id_comuna) 
        REFERENCES comunas(id_comuna) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 5. TABLA: PERFILES_CLIENTE
-- ----------------------------------------------------------------------------
CREATE TABLE perfiles_cliente (
    id_perfil INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL UNIQUE,
    descripcion_tipo_consumo VARCHAR(100) NOT NULL DEFAULT 'Residencial (Hogar)',
    descripcion_horario_entrega VARCHAR(100) NOT NULL DEFAULT 'Mañana (09:00 - 13:00)',
    descripcion_tipo_valvula VARCHAR(100) NOT NULL DEFAULT 'Válvula Tradicional (Clip)',
    descripcion_condicion_envase VARCHAR(100) NOT NULL DEFAULT 'Con intercambio de cilindro vacío',
    telefono_contacto VARCHAR(20) NULL,
    CONSTRAINT fk_perfiles_usuarios FOREIGN KEY (id_usuario) 
        REFERENCES usuarios(id_usuario) ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 6. TABLAS: MARCAS Y CATEGORIAS
-- ----------------------------------------------------------------------------
CREATE TABLE marcas (
    id_marca INT AUTO_INCREMENT PRIMARY KEY,
    nombre_marca VARCHAR(50) NOT NULL UNIQUE
) ENGINE=InnoDB;

CREATE TABLE categorias (
    id_categoria INT AUTO_INCREMENT PRIMARY KEY,
    nombre_categoria VARCHAR(50) NOT NULL UNIQUE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 7. TABLA: PRODUCTOS
-- ----------------------------------------------------------------------------
CREATE TABLE productos (
    id_producto INT AUTO_INCREMENT PRIMARY KEY,
    id_marca INT NOT NULL,
    id_categoria INT NOT NULL,
    nombre_producto VARCHAR(120) NOT NULL,
    descripcion_producto TEXT NULL,
    precio INT NOT NULL CHECK (precio > 0),
    imagen_url VARCHAR(255) DEFAULT '/img/logo.png',
    stock_disponible INT NOT NULL DEFAULT 50 CHECK (stock_disponible >= 0),
    es_personalizado BOOLEAN DEFAULT FALSE,
    producto_activo BOOLEAN DEFAULT TRUE,
    CONSTRAINT fk_productos_marcas FOREIGN KEY (id_marca) 
        REFERENCES marcas(id_marca) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_productos_categorias FOREIGN KEY (id_categoria) 
        REFERENCES categorias(id_categoria) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 8. TABLA: ITEMS_CARRITO_SESION (Acumulador temporal previo a la compra)
-- Representa el subtotal acumulado dinámicamente en el catálogo al seleccionar balones
-- ----------------------------------------------------------------------------
CREATE TABLE items_carrito_sesion (
    id_item_carrito INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    id_producto INT NOT NULL,
    cantidad INT NOT NULL DEFAULT 1 CHECK (cantidad > 0),
    fecha_agregado TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_carrito_usuarios FOREIGN KEY (id_usuario) 
        REFERENCES usuarios(id_usuario) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_carrito_productos FOREIGN KEY (id_producto) 
        REFERENCES productos(id_producto) ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 9. TABLA: PEDIDOS (Cabecera de compra confirmada)
-- ----------------------------------------------------------------------------
CREATE TABLE pedidos (
    id_pedido INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    fecha_pedido TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    importe_total_pedido INT NOT NULL CHECK (importe_total_pedido > 0),
    descripcion_estado_pedido VARCHAR(50) NOT NULL DEFAULT 'Pendiente de Despacho',
    direccion_despacho_historica VARCHAR(200) NOT NULL,
    CONSTRAINT fk_pedidos_usuarios FOREIGN KEY (id_usuario) 
        REFERENCES usuarios(id_usuario) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 10. TABLA: DETALLES_PEDIDO
-- El subtotal_linea es una columna generada de forma matemática automática,
-- impidiendo la inserción de valores arbitrarios a mano.
-- ----------------------------------------------------------------------------
CREATE TABLE detalles_pedido (
    id_detalle INT AUTO_INCREMENT PRIMARY KEY,
    id_pedido INT NOT NULL,
    id_producto INT NOT NULL,
    cantidad_unidades INT NOT NULL CHECK (cantidad_unidades > 0),
    precio_unitario_venta INT NOT NULL CHECK (precio_unitario_venta > 0),
    subtotal_linea INT GENERATED ALWAYS AS (cantidad_unidades * precio_unitario_venta) STORED,
    CONSTRAINT fk_detalles_pedidos FOREIGN KEY (id_pedido) 
        REFERENCES pedidos(id_pedido) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_detalles_productos FOREIGN KEY (id_producto) 
        REFERENCES productos(id_producto) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 11. TABLA: FAVORITOS
-- ----------------------------------------------------------------------------
CREATE TABLE favoritos (
    id_usuario INT NOT NULL,
    id_producto INT NOT NULL,
    fecha_agregado TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id_usuario, id_producto),
    CONSTRAINT fk_favoritos_usuarios FOREIGN KEY (id_usuario) 
        REFERENCES usuarios(id_usuario) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_favoritos_productos FOREIGN KEY (id_producto) 
        REFERENCES productos(id_producto) ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;
