-- ============================================================================
-- BASE DE DATOS: Gas Licuado "El Volcán"
-- Compatibilidad: MySQL / MariaDB (XAMPP) y Spring Boot (JPA / Hibernate)
-- ============================================================================

DROP DATABASE IF EXISTS el_volcan_db;
CREATE DATABASE el_volcan_db CHARACTER SET utf8mb4 COLLATE utf8mb4_spanish_ci;
USE el_volcan_db;

-- ----------------------------------------------------------------------------
-- 1. TABLA: ROLES
-- ----------------------------------------------------------------------------
CREATE TABLE roles (
    id_rol INT AUTO_INCREMENT PRIMARY KEY,
    nombre_rol VARCHAR(30) NOT NULL UNIQUE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 2. TABLA: USUARIOS
-- ----------------------------------------------------------------------------
CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    id_rol INT NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL, -- Longitud pensada para BCrypt en Spring Boot
    nombre VARCHAR(60) NOT NULL,
    apellido VARCHAR(60) NOT NULL,
    fecha_nacimiento DATE NOT NULL,
    activo BOOLEAN DEFAULT TRUE,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_usuarios_roles FOREIGN KEY (id_rol) 
        REFERENCES roles(id_rol) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 3. TABLA: PERFILES_CLIENTE (Campos de contexto logístico del negocio)
-- ----------------------------------------------------------------------------
CREATE TABLE perfiles_cliente (
    id_perfil INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL UNIQUE,
    direccion_despacho VARCHAR(200) NOT NULL,
    tipo_consumo ENUM('Residencial (Hogar)', 'Comercial (Restaurante / Negocio)', 'Industrial') NOT NULL DEFAULT 'Residencial (Hogar)',
    horario_preferente ENUM('Mañana (09:00 - 13:00)', 'Tarde (14:00 - 18:00)', 'Vespertino (18:00 - 21:00)') NOT NULL DEFAULT 'Mañana (09:00 - 13:00)',
    tipo_valvula ENUM('Válvula Tradicional (Clip)', 'Válvula de Rosca Rápida', 'No estoy seguro (solicitar revisión técnica)') NOT NULL DEFAULT 'Válvula Tradicional (Clip)',
    tiene_envase ENUM('Sí, tengo envase para intercambio', 'No tengo envase (requiero comprar cilindro nuevo)') NOT NULL DEFAULT 'Sí, tengo envase para intercambio',
    telefono VARCHAR(20) NULL,
    CONSTRAINT fk_perfiles_usuarios FOREIGN KEY (id_usuario) 
        REFERENCES usuarios(id_usuario) ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 4. TABLA: MARCAS
-- ----------------------------------------------------------------------------
CREATE TABLE marcas (
    id_marca INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 5. TABLA: CATEGORIAS
-- ----------------------------------------------------------------------------
CREATE TABLE categorias (
    id_categoria INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 6. TABLA: PRODUCTOS (Con restricción de precio > 0)
-- ----------------------------------------------------------------------------
CREATE TABLE productos (
    id_producto INT AUTO_INCREMENT PRIMARY KEY,
    id_marca INT NOT NULL,
    id_categoria INT NOT NULL,
    nombre VARCHAR(120) NOT NULL,
    descripcion TEXT NULL,
    precio INT NOT NULL,
    imagen_url VARCHAR(255) DEFAULT '/img/logo.png',
    stock INT NOT NULL DEFAULT 50,
    es_personalizado BOOLEAN DEFAULT FALSE,
    activo BOOLEAN DEFAULT TRUE,
    CONSTRAINT chk_precio_positivo CHECK (precio > 0),
    CONSTRAINT chk_stock_no_negativo CHECK (stock >= 0),
    CONSTRAINT fk_productos_marcas FOREIGN KEY (id_marca) 
        REFERENCES marcas(id_marca) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_productos_categorias FOREIGN KEY (id_categoria) 
        REFERENCES categorias(id_categoria) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 7. TABLA: PEDIDOS (Cabecera)
-- ----------------------------------------------------------------------------
CREATE TABLE pedidos (
    id_pedido INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    fecha_pedido TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    total INT NOT NULL,
    estado_pedido ENUM('PENDIENTE', 'EN_CAMINO', 'ENTREGADO', 'CANCELADO') NOT NULL DEFAULT 'PENDIENTE',
    direccion_despacho_registro VARCHAR(200) NOT NULL,
    CONSTRAINT chk_total_positivo CHECK (total > 0),
    CONSTRAINT fk_pedidos_usuarios FOREIGN KEY (id_usuario) 
        REFERENCES usuarios(id_usuario) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 8. TABLA: DETALLES_PEDIDO (Cuerpo / Carrito comprado)
-- ----------------------------------------------------------------------------
CREATE TABLE detalles_pedido (
    id_detalle INT AUTO_INCREMENT PRIMARY KEY,
    id_pedido INT NOT NULL,
    id_producto INT NOT NULL,
    cantidad INT NOT NULL,
    precio_unitario INT NOT NULL,
    subtotal INT NOT NULL,
    CONSTRAINT chk_cantidad_positiva CHECK (cantidad > 0),
    CONSTRAINT fk_detalles_pedidos FOREIGN KEY (id_pedido) 
        REFERENCES pedidos(id_pedido) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_detalles_productos FOREIGN KEY (id_producto) 
        REFERENCES productos(id_producto) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 9. TABLA: FAVORITOS (Tabla asociativa N a M)
-- ----------------------------------------------------------------------------
CREATE TABLE favoritos (
    id_usuario INT NOT NULL,
    id_producto INT NOT NULL,
    fecha_agregado TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id_usuario, id_producto),
    CONSTRAINT fk_favoritos_usuarios FOREIGN KEY (id_usuario) 
        REFERENCES usuarios(id_usuario) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_favoritos_productos FOREIGN KEY (id_producto) 
        REFERENCES productos(id_producto) ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;
